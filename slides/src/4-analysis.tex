\section{Analisi}
\subsection{Parallel n-grams}

\begin{frame}{Analisi}
    Confronto delle prestazioni tra versione sequenziale e parallela su dataset di diverse dimensioni.
    \begin{itemize}
        \item Dataset: da 100KB a 2 GB.
        \item Metrica: Speed-up $T_{seq} / T_{par}$.
    \end{itemize}
\end{frame}

\begin{frame}{Analisi}
    \begin{block}{Testo}
        Esempi di 3-grams più frequenti trovati:
        \begin{itemize}
            \item \textbf{100KB}: This is the unique input size for which the most frequent 3-grams are \textit{randallsville new york} (24), \textit{variety of quartz} (10) and \textit{in the uk} (9).
            \item \textbf{100MB - 2GB:} The most frequent 3-grams always the same: \textit{one of the}, \textit{as well as}, and \textit{a lot of}. The frequencies in 2 GB file are respectively: 157788, 142530, and 86345.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Analisi}
    \begin{block}{Tempo - 100KB}
        Parallelo più lento per l'overhead del fork-join.
        \begin{itemize}
            \item Seq: 0.02s.
            \item Par: 0.057s.
        \end{itemize}
    \end{block}
\end{frame}
