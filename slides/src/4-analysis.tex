\section{Analisi}
\subsection{Parallel n-grams}

\begin{frame}{Analisi}
    Confronto delle prestazioni tra versione sequenziale e parallela su dataset di diverse dimensioni.
    \begin{itemize}
        \item Dataset: da 100KB a 1.5 GB.
        \item Metrica: Speed-up $T_{seq} / T_{par}$.
    \end{itemize}
\end{frame}

\begin{frame}{Analisi}
    \begin{block}{Testo}
        Esempi di 3-grams più frequenti trovati:
        \begin{itemize}
            \item \textbf{100KB}: This is the unique input size for which the most frequent 3-grams are \textit{randallsville new york} (24), \textit{variety of quartz} (10) and \textit{in the uk} (9).
            \item \textbf{100MB - 1.5GB:} The most frequent 3-grams always the same: \textit{one of the}, \textit{as well as}, and \textit{a lot of}. The frequencies in 1.5 GB file are respectively: 118445, 106364, and 64775.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Analisi}
    \begin{block}{Hash Table}
        Fill factor $\alpha=\tfrac{busy\_buckets}{table\_length}$:
        \begin{itemize}
            \item \textbf{100 KB}: 0.0017.
            \item \textbf{100 MB}: 0.6853.
            \item \textbf{500 MB}: 0.991.
            \item \textbf{1 GB}: 0.9981.
            \item \textbf{1.5 GB:} 1.0000.
        \end{itemize}
    \end{block}
\end{frame}
